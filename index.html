-- ============================================================ -- БРОНИ: завершение посещения ("гости ушли") отдельно от отмены. -- Новый статус 'finished' = гости были и ушли (история сохраняется). -- 'cancelled'/'no_show' = бронь сорвалась. -- Плюс автозавершение «засидевшихся» столов после 05:00. -- Выполнять в SQL Editor. Безопасно для повторного запуска. -- ============================================================ -- 1) Разрешаем новый статус 'finished' alter table bookings drop constraint if exists bookings_status_check; alter table bookings add constraint bookings_status_check check (status in ('booked','seated','cancelled','no_show','finished')); -- 2) Кто и когда отметил «гости ушли» alter table bookings add column if not exists finished_by_name text; alter table bookings add column if not exists finished_at timestamptz; -- 3) Счётчик визитов гостя: считаем и состоявшиеся посадки, и завершённые -- (раньше считалось только 'seated'; теперь и 'finished' — гость реально был) drop view if exists v_bookings_full; create view v_bookings_full as select b.*, t.code as table_code, t.name as table_name, t.kind as table_kind, z.key as zone_key, z.title as zone_title, v.key as venue_key, coalesce(g.name, b.guest_name) as display_name, coalesce(b.phone, g.phone_norm) as display_phone, coalesce(g.visits_base, 0) + (select count(*) from bookings b2 where b2.guest_id = b.guest_id and b2.status in ('seated','finished') and b2.id <> b.id) as visit_count from bookings b join tables t on t.id = b.table_id join zones z on z.id = t.zone_id join venues v on v.id = z.venue_id left join guests g on g.id = b.guest_id; -- 4) Автозавершение: брони со статусом 'seated', чья бизнес-дата уже прошла -- (т.е. рабочий день закрыт в 05:00), помечаем 'finished' автоматически. -- Вызывается с клиента при загрузке; помечает только «вчера и ранее». create or replace function auto_finish_seated() returns integer language plpgsql security definer as $$ declare n integer; begin update bookings set status = 'finished', finished_by_name = coalesce(finished_by_name, 'авто (после 05:00)'), finished_at = coalesce(finished_at, now()) where status = 'seated' and business_date < business_date_of(now()); get diagnostics n = row_count; return n; end; $$; grant execute on function auto_finish_seated() to anon, authenticated;
